matrixManipulation := proc(A)
    local AA, opt, input, row1, row2, scalar;

    # Check if the input matrix AA is valid
    if not type(A, 'Matrix') then
        error "The input must be a valid matrix.";
    end if;
	
	AA := copy(A);
    while true do
        # Display the current state of the matrix
        print("Current matrix AA:");
        print(eval(AA));

        # Prompt the user for the operation they wish to perform
        lprint("[1] Swap two rows [2] Replace a row with a multiple of another row [3] Exit");
        opt := parse(readline()):

        if opt = 1 then
            # Swap two rows
            lprint("Enter the two rows to swap");
            row1 := parse(readline()):
			#print("Swap:", row1);
            row2 := parse(readline()):
			#print("Swap:", row1, row2);
            LinearAlgebra[RowOperation](AA, [row1, row2], inplace=true);

        elif opt = 2 then
            # Replace a row with a multiple of another row
            print("Enter the target row and the source row with its scalar multiplier (e.g., 2 1 3):");
            row1, row2, scalar := op([parse(ReadLine())]):
            AA := RowOperation(AA, ["add", row1, scalar, row2]);

        elif opt = 3 then
            # Exit the loop
            print("Exiting the procedure. Final matrix AA:");
            print(AA);
            break;

        else
            # Invalid opt
            print("Invalid opt. Please try again.");
        end if;
    end do;

end proc;
